# powershell.exe -Command (not -File) is required: -File mode collapses
# empty quoted arguments before parameter binding, which breaks every
# upload menu choice that leaves uf2_volume_label / uf2_model / uf2_board_id
# blank. -Command preserves '' as an empty string token.
tools.niusopenocd.upload.pattern.windows=powershell -NoProfile -ExecutionPolicy Bypass -Command "& '{runtime.platform.path}/tools/niusrobotlab/upload.ps1' -Mode openocd -Tool '{runtime.tools.openocd-0.11.0-arduino2.path}/bin/openocd.exe' -ScriptRoot '{runtime.tools.openocd-0.11.0-arduino2.path}/share/openocd/scripts' -Config '{runtime.platform.path}/tools/openocd/{build.openocdscript}' -Hex '{build.path}/{build.project_name}.hex' -Board '{build.variant}' {upload.verbose}"

# debug.server.usbgdbstub.* — unused at the arduino-cli level. arduino-cli's
# Debug command only accepts a fixed set of server-type names (openocd,
# jlink, stutil, pyocd, bmp); a custom value here would be rejected with
# "GDB server 'usbgdbstub' is not supported". The buildprofile=usbgdbstub
# menu in boards.txt instead keeps `debug.server=openocd` and overrides
# `debug.server.openocd.path` to point at usb_gdbstub_server.cmd, which
# spawns usb_gdbstub_bridge.ps1. The bridge prints
# "Listening on port 3335 for gdb connections" right after its TCP socket
# is ready — matching cortex-debug's default openocd serverReady regex,
# so cortex-debug spawns it as the debug server and then connects GDB to
# localhost:3335. One-click Debug in Arduino IDE 2 with no separate
# "start bridge" terminal.
#
# These keys are left in place (empty) so the platform reads cleanly with
# arduino-cli's verbose debug --info even when usbgdbstub-specific lookups
# fall back here. The bridge.{describe,listports,launch}.pattern entries
# above remain available for manual command-line debugging.

debug.server.usbgdbstub.scripts_dir=
debug.server.usbgdbstub.script=
